Generate this file on any machine with:

```bash
nobackups init -c ./config.yaml
```

Option-by-option docs: [global settings](/configuration/overview#global-settings), [destinations](/configuration/destinations), [jobs](/configuration/jobs), [encryption](/configuration/encryption), [retention](/configuration/retention), [hooks](/configuration/hooks), [notifications](/configuration/notifications).

```yaml config.yaml
# NoBackups configuration
#
# Any value can reference environment variables as ${VAR} or ${VAR:-default}.
# Variables are also loaded from nobackups.env next to this file, so keep
# secrets there (chmod 600) instead of in this file.

# Where locks and last-run status files are kept.
state_dir: /var/lib/nobackups
# debug | info | warn | error
log_level: info
# Defaults to the machine's hostname. Usable as {hostname} in prefixes.
# hostname: web01

# ---------------------------------------------------------------------------
# Destinations: where backups go. Give each one a name, then reference it
# from jobs. Every S3-compatible store uses type: s3.
# ---------------------------------------------------------------------------
destinations:
  # type is s3, local, or a provider shorthand that fills in the provider's
  # defaults: alarik, rustfs, minio, garage, hetzner, aws, r2, b2,
  # digitalocean, wasabi. Anything you set yourself overrides the defaults.

  # Hetzner Object Storage: region is fsn1, nbg1 or hel1; the endpoint is
  # derived from it.
  hetzner:
    type: hetzner
    region: fsn1
    bucket: my-backups
    access_key_id: ${HETZNER_ACCESS_KEY}
    secret_access_key: ${HETZNER_SECRET_KEY}
    prefix: servers/{hostname}

  # Self-hosted Alarik / RustFS / MinIO / Garage (path-style by default).
  # Use http:// for plain HTTP.
  # alarik:
  #   type: alarik
  #   endpoint: s3.example.com
  #   bucket: backups
  #   access_key_id: ${ALARIK_ACCESS_KEY}
  #   secret_access_key: ${ALARIK_SECRET_KEY}
  #   prefix: "{hostname}"
  #   # ca_file: /etc/nobackups/alarik-ca.pem   # private CA for HTTPS
  #   # insecure_skip_verify: false            # last resort for self-signed certs

  # AWS S3 (endpoint derived from region)
  # aws:
  #   type: aws
  #   region: eu-central-1
  #   bucket: my-backups
  #   access_key_id: ${AWS_ACCESS_KEY_ID}
  #   secret_access_key: ${AWS_SECRET_ACCESS_KEY}
  #   storage_class: STANDARD_IA

  # Any other S3-compatible service: type s3 with an explicit endpoint.
  # other:
  #   type: s3
  #   endpoint: s3.example.net
  #   region: us-east-1
  #   bucket: my-backups
  #   access_key_id: ${S3_ACCESS_KEY}
  #   secret_access_key: ${S3_SECRET_KEY}

  # A local directory, e.g. a second disk or NFS mount.
  # local:
  #   type: local
  #   path: /mnt/backup
  #   prefix: "{hostname}"

  # Optional for any s3 destination:
  #   part_size_mb: 64   # multipart chunk size; max object size = 10000 x this.
  #                      # Memory use is roughly part_size_mb per destination.

# ---------------------------------------------------------------------------
# Jobs: what to back up, when, and where.
# ---------------------------------------------------------------------------
jobs:
  - name: system
    sources:
      - /etc
      - /root
      - /home
    # Patterns without "/" match file/dir names anywhere; patterns with "/"
    # match full paths (and everything below them).
    exclude:
      - "*.tmp"
      - "*.swp"
      - ".cache"
      - "node_modules"
      - /home/*/Downloads
    destinations: [hetzner]
    # Cron (minute hour day month weekday) or @hourly / @daily / "@every 6h".
    # Omit to only run manually with `nobackups run system`.
    schedule: "15 3 * * *"
    # zstd (default) | gzip | none
    compression: zstd
    # compression_level: 3
    # Don't descend into other mounted filesystems (e.g. /proc, NFS mounts).
    one_file_system: true
    # Encrypt with age before upload. Use ONE of passphrase, passphrase_file
    # or recipients (age public keys). Lose the key = lose the backups.
    encryption:
      passphrase: ${BACKUP_PASSPHRASE}
    # Keep a snapshot if it is one of the newest keep_last OR younger than
    # keep_days. Omit both to keep everything.
    retention:
      keep_last: 7
      keep_days: 30
    # Abort the run if it takes longer than this.
    timeout: 6h
    # Optional per-job notifications, sent in addition to the global ones
    # below. Events: success, warning (some files unreadable), failure.
    # notify:
    #   discord:
    #     - url: ${DISCORD_WEBHOOK_URL}
    #       on: [failure, warning]      # default
    #       mention: "<@&123456789>"    # optional role/user ping on failure/warning

  # Databases are dumped with their own tools (pg_dump, mysqldump, mongodump,
  # redis-cli, sqlite3) and stored in the snapshot under nobackups-databases/.
  # Set container to dump from inside a running Docker container (a unique
  # name prefix is enough, e.g. Swarm's "name.1.abc"); leave it out to use
  # the tools installed on this server.
  # - name: databases
  #   databases:
  #     - type: postgres            # postgres | mysql | mariadb | mongodb | redis | sqlite
  #       container: my-postgres
  #       user: postgres
  #       password: ${PG_PASSWORD}
  #       database: app             # leave out to dump all databases
  #     - type: mariadb
  #       host: 127.0.0.1
  #       user: backup
  #       password: ${MYSQL_PASSWORD}
  #     - type: sqlite
  #       path: /opt/vaultwarden/data/db.sqlite3
  #   destinations: [hetzner]
  #   schedule: "0 */6 * * *"
  #   encryption:
  #     recipients:
  #       - age1qyqszqgpqyqszqgpqyqszqgpqyqszqgpqyqszqgpqyqszqgpqyqs3290gq
  #     # Private key, only needed on the machine you restore from:
  #     # identity_file: /root/backup-key.txt
  #   retention:
  #     keep_last: 28
  #
  # Hooks run shell commands before and after a job, e.g. to stop a service
  # while its files are copied. after hooks always run and get
  # $NOBACKUPS_STATUS (success or failure).
  # - name: app-files
  #   sources: [/srv/app/uploads]
  #   destinations: [hetzner]
  #   hooks:
  #     before: [systemctl stop app-worker]
  #     after: [systemctl start app-worker]

# ---------------------------------------------------------------------------
# Notifications for every job. Jobs can add their own under `notify:` too.
# Events: success | warning (backup ok, some files unreadable) | failure.
# Subscribing to success also delivers warnings.
# Test with: nobackups test-notify [job] --event failure
# ---------------------------------------------------------------------------
# notify:
#   # Discord: rich embed with status colour, host, size, duration,
#   # per-destination results and the error message.
#   discord:
#     - url: ${DISCORD_WEBHOOK_URL}   # Server Settings → Integrations → Webhooks
#       on: [failure, warning]       # default; add success for every run
#       mention: "@here"             # optional; only used for failure/warning
#       # username: Backups          # default: NoBackups
#
#   # Generic JSON webhooks (Slack, Mattermost, ntfy, n8n, healthchecks.io...).
#   # The payload has event, text and the full result. Default on: [failure].
#   webhooks:
#     - url: https://hooks.slack.com/services/XXX
#       on: [failure]
```
